### The Script: `rename_jpeg_to_jpg.sh`

#!/bin/bash

# 1. Define the target directory (default to current directory if not provided)
TARGET_DIR="${1:-.}"

# 2. Check if the target directory exists
if [ ! -d "$TARGET_DIR" ]; then
    echo "Error: Directory '$TARGET_DIR' does not exist."
    exit 1
fi

# 3. Find all .jpeg files (case-insensitive) and rename them to .jpg
#    -type f: Only match regular files
#    -iname "*.jpeg": Case-insensitive match for .jpeg extension
